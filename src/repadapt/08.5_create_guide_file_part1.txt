# Define important file locations

# WORKING_FOLDER is the core folder where this pipeline is being run.
WORKING_FOLDER=/gpfs2/scratch/elongman/Nucella_can_Seascape

# This is the location where the reference genome from the first round of pilon and all its indexes are stored.
REFERENCE=$WORKING_FOLDER/data/processed/repadapt/genome/N.canaliculata_assembly.fasta.softmasked.fa

#--------------------------------------------------------------------------------

# Change directory
cd $WORKING_FOLDER/data/processed/repadapt/guide_files

# Get all of the contig names 
grep ">" $REFERENCE > chromosomes.txt

# Get rid of the ">"
sed -i 's/>//g' chromosomes.txt

#--------------------------------------------------------------------------------

# Change directory
cd $WORKING_FOLDER/data/processed/repadapt

# Make list of ALL the realigned bam files
ls $WORKING_FOLDER/data/processed/repadapt/realign_indel_fix_RG/*realigned.bam > samples.list.txt
